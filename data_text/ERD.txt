Table users {
  id UUID [pk, default: `gen_random_uuid()`]
  full_name VARCHAR(100) [not null]
  user_code VARCHAR(50) [not null, unique, note: 'MSNV or MSSV']
  role VARCHAR(50) [not null, note: 'ADMIN | FACILITY_MANAGER | GUARD | NORMAL_USER']
  email VARCHAR(255) [not null, unique]
  password VARCHAR(60) [note: 'b64-encoded bcrypt hash, length=60'] 
  is_active BOOLEAN [not null, default: true, note: 'true: Active, false: Inactive']
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ [note: 'Inactive = soft-delete']
  import_batch_id UUID
}

Table password_reset_tokens {
  id UUID [pk, default: `gen_random_uuid()`]
  token VARCHAR(255) [not null, unique]
  email VARCHAR(255) [not null]
  expired_at TIMESTAMPTZ [not null]
  is_used BOOLEAN [not null, default: false]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

Table areas {
  id UUID [pk, default: `gen_random_uuid()`]
  code VARCHAR(50) [not null]
  name VARCHAR(150) [not null]
  area_level VARCHAR(30) [not null, note: 'PUBLIC | SEMI_PRIVATE | PRIVATE']
  building VARCHAR(50)
  floor VARCHAR(20)
  description TEXT
  geometry JSONB [note: 'Area polygon geometry (type, version, vertices)']
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ
}

Table floor_plans {
  id UUID [pk, default: `gen_random_uuid()`]
  building VARCHAR(50) [not null]
  floor VARCHAR(20) [not null]
  image_key VARCHAR(255) [not null]
  original_width INTEGER [not null]
  original_height INTEGER [not null]
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  
  indexes {
    (`LOWER(building)`, `LOWER(floor)`) [unique, name: 'uq_floor_plans_building_floor']
  }
}

Table cameras {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_code VARCHAR(50) [not null, unique, note: 'CAM-001']
  name VARCHAR(100) [not null, note: 'Tên camera']
  status VARCHAR(50) [not null, note: 'ACTIVE | DECOMMISSIONED']
  operational_status VARCHAR(50) [not null, note: 'ONLINE | OFFLINE']
  installed_at TIMESTAMPTZ
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ
}

Table camera_stream_configurations {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, unique, note: 'FK to cameras']
  host VARCHAR(255) [not null]
  port INT [not null]
  username VARCHAR(100)
  credential_ref VARCHAR(255)
  main_stream_path VARCHAR(512) [not null]
  sub_stream_path VARCHAR(512)
  retries_before_alert INT [not null, default: 3]
  timeout_ms INT [not null, default: 5000]
}

Table ai_configurations {
  id UUID [pk, default: `gen_random_uuid()`]
  face_match_threshold NUMERIC(3,2) [not null, note: 'ngưỡng tin cậy nhận dạng khuôn mặt']
  inference_fps INT [not null, note: 'số frame AI xử lý trong 1s']
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

Table face_data {
  id UUID [pk, default: `gen_random_uuid()`]
  user_id UUID [not null, unique, note: 'FK to users']
  code VARCHAR(50) [not null, note: 'MSSV hoặc MSNV']
  image_front_url VARCHAR(512) [not null]
  embedding_front vector(512) [not null]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

Table camera_health_logs {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, note: 'FK to cameras']
  status VARCHAR(50) [not null, note: 'ONLINE | OFFLINE | ERROR']
  checked_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  latency_ms INT [note: 'Độ trễ camera phản hồi tính bằng mili giây']
  fps NUMERIC(5,2) [note: 'Số frame/s thực tế']
  error_code VARCHAR(100) [note: 'Mã lỗi camera']
  error_message VARCHAR(255) [note: 'Mô tả lỗi camera']
}

// Bảng trung gian Area N-N Camera
Table area_cameras {
  area_id UUID [not null, note: 'FK to areas']
  camera_id UUID [not null, note: 'FK to cameras']
  
  indexes {
    (area_id, camera_id) [pk]
  }
}

Table access_requests {
  id UUID [pk, default: `gen_random_uuid()`]
  area_id UUID [not null, note: 'FK to areas — khu vực cần truy cập']
  requester_id UUID [not null, note: 'FK to users — người tạo request (cá nhân hoặc đại diện nhóm)']
  request_type VARCHAR(20) [not null, note: 'INDIVIDUAL | GROUP']
  purpose TEXT [not null, note: 'Mục đích sử dụng']
  start_time TIMESTAMPTZ [not null, note: 'Thời gian bắt đầu sử dụng']
  end_time TIMESTAMPTZ [not null, note: 'Thời gian kết thúc sử dụng']
  status VARCHAR(20) [not null, default: 'PENDING', note: 'PENDING | APPROVED | REJECTED | CANCELLED | EXPIRED']
  reviewer_id UUID [note: 'FK to users — FM phê duyệt']
  reviewed_at TIMESTAMPTZ [note: 'Thời điểm FM xử lý']
  rejection_reason TEXT [note: 'Lý do từ chối nếu bị reject']
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

Table access_request_members {
  id UUID [pk, default: `gen_random_uuid()`]
  access_request_id UUID [not null, note: 'FK to access_requests']
  user_id UUID [not null, note: 'FK to users — thành viên trong nhóm được yêu cầu truy cập']

  indexes {
    (access_request_id, user_id) [unique]
  }
}

Table notifications {
  id UUID [pk, default: `gen_random_uuid()`]
  recipient_id UUID [not null, note: 'FK to users']
  type VARCHAR(40) [not null, note: 'REQUEST_APPROVED | REQUEST_REJECTED | EXPIRING_SOON | ACCESS_DENIED | ADDED_TO_GROUP | NEW_REQUEST_PENDING | REQUEST_CANCELLED | PENDING_OVERDUE']
  title VARCHAR(255) [not null]
  message TEXT [not null]
  reference_id UUID
  reference_type VARCHAR(40)
  is_read BOOLEAN [not null, default: false]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  read_at TIMESTAMPTZ
}

Table system_configurations {
  config_key VARCHAR(100) [pk]
  config_value VARCHAR(255) [not null]
  data_type VARCHAR(20) [not null, note: 'INTEGER | DECIMAL | BOOLEAN | STRING']
  min_value NUMERIC
  max_value NUMERIC
  unit VARCHAR(30)
  config_group VARCHAR(50) [not null]
  display_order INT [not null]
  description TEXT [not null]
  editable BOOLEAN [not null, default: true]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_by UUID [note: 'FK to users']
}

Table system_configuration_change_logs {
  id UUID [pk, default: `gen_random_uuid()`]
  config_key VARCHAR(100) [not null]
  old_value VARCHAR(255)
  new_value VARCHAR(255) [not null]
  changed_by UUID [note: 'FK to users']
  changed_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

Table guard_schedule_templates {
  id UUID [pk, default: `gen_random_uuid()`]
  guard_id UUID [not null, note: 'FK to users (role GUARD)']
  day_of_week INT [not null, note: '1: Sunday, 2: Monday, ..., 7: Saturday']
  shift_type VARCHAR(20) [not null, note: 'SHIFT_MORNING | SHIFT_AFTERNOON | SHIFT_NIGHT | SHIFT_OFF']
  start_time TIME [not null]
  end_time TIME [not null]
  area_id UUID [note: 'FK to areas — chốt trực cụ thể (Phòng bảo vệ, Sảnh, Tuần tra)']
  radio_channel VARCHAR(50) [note: 'Kênh bộ đàm, vd: Kênh 2 - Tòa Alpha']
  notes TEXT [note: 'Ghi chú nhiệm vụ']
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

Table guard_shifts {
  id UUID [pk, default: `gen_random_uuid()`]
  guard_id UUID [not null, note: 'FK to users']
  shift_date DATE [not null, note: 'Ngày trực cụ thể YYYY-MM-DD']
  shift_type VARCHAR(20) [not null]
  start_time TIME [not null]
  end_time TIME [not null]
  area_id UUID [note: 'FK to areas']
  radio_channel VARCHAR(50)
  status VARCHAR(20) [not null, default: 'SCHEDULED', note: 'SCHEDULED | CHECKED_IN | COMPLETED | ABSENT | CANCELLED']
  check_in_at TIMESTAMPTZ
  check_out_at TIMESTAMPTZ
  notes TEXT
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]

  indexes {
    (guard_id, shift_date, start_time) [unique]
    (shift_date, area_id)
  }
}

Table security_incidents {
  id UUID [pk, default: `gen_random_uuid()`]
  event_id VARCHAR(100) [note: 'ID sự kiện từ AI Service Kafka']
  camera_code VARCHAR(50) [not null, note: 'Mã camera phát hiện']
  area_id UUID [not null, note: 'FK to areas']
  building VARCHAR(50) [not null, note: 'Mã tòa nhà / cơ sở để lọc thông báo tức thì']
  event_type VARCHAR(50) [not null, note: 'UNAUTHORIZED_ENTRY | UNKNOWN_PERSON | AFTER_HOURS_ACCESS | LOITERING | CROWD_OVERCROWDING']
  image_url VARCHAR(512) [note: 'Ảnh snapshot bằng chứng từ MinIO']
  detected_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  status VARCHAR(30) [not null, default: 'NEW', note: 'NEW | CLAIMED | RESOLVED_VERIFIED | RESOLVED_DISMISSED']
  
  // Thông tin tiếp nhận xử lý
  claimed_by UUID [note: 'FK to users — bảo vệ nhận việc trên mobile app']
  claimed_at TIMESTAMPTZ
  
  // Báo cáo kết quả xử lý
  resolved_by UUID [note: 'FK to users — bảo vệ hoàn tất xử lý']
  resolved_at TIMESTAMPTZ
  outcome VARCHAR(20) [note: 'VERIFIED | DISMISSED']
  resolution_category VARCHAR(50) [note: 'FALSE_ALARM | AUTHORIZED_EXCEPTION | REMINDED_DISPERSED | ESCORTED_OUT | REPORT_FILED | DETAINED_ESCALATED | OTHER']
  resolution_notes TEXT [note: 'Ghi chú biên bản xử lý của bảo vệ']
  evidence_image_url VARCHAR(512) [note: 'Ảnh hiện trường sau xử lý nếu có']
  version INT [not null, default: 0, note: 'Optimistic locking counter chống tranh chấp khi claim']

  indexes {
    (building, status)
    (detected_at)
  }
}

// ============================================================================
// ENTITY OF AUTHENTICATION MODULE
// ============================================================================
Table users {
  id UUID [pk, default: `gen_random_uuid()`]
  full_name VARCHAR(100) [not null]
  user_code VARCHAR(50) [not null, unique, note: 'MSNV or MSSV']
  role VARCHAR(50) [not null, note: 'ADMIN | FACILITY_MANAGER | INTERNAL_GUARD | OUTSOURCED_GUARD | NORMAL_USER']
  email VARCHAR(255) [not null, unique]
  password VARCHAR(60) [note: 'b64-encoded bcrypt hash, length=60'] 
  is_active BOOLEAN [not null, default: true, note: 'true: Active, false: Inactive']
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ [note: 'Inactive = soft-delete']
}

Table password_reset_tokens {
  id UUID [pk, default: `gen_random_uuid()`]
  token VARCHAR(255) [not null, unique]
  email VARCHAR(255) [not null]
  expired_at TIMESTAMPTZ [not null]
  is_used BOOLEAN [not null, default: false]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

// ============================================================================
// ENTITY OF M06 AREA MANAGEMENT MODULE
// ============================================================================
Table areas {
  id UUID [pk, default: `gen_random_uuid()`]
  code VARCHAR(50) [not null]
  name VARCHAR(150) [not null]
  area_level VARCHAR(30) [not null, note: 'PUBLIC | SEMI_PRIVATE | PRIVATE']
  building VARCHAR(50)
  floor VARCHAR(20)
  description TEXT
  geometry JSONB [note: 'Area polygon geometry (type, version, vertices)']
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ
}

Table floor_plans {
  id UUID [pk, default: `gen_random_uuid()`]
  building VARCHAR(50) [not null]
  floor VARCHAR(20) [not null]
  image_key VARCHAR(255) [not null]
  original_width INTEGER [not null]
  original_height INTEGER [not null]
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  
  indexes {
    (`LOWER(building)`, `LOWER(floor)`) [unique, name: 'uq_floor_plans_building_floor']
  }
}

// ============================================================================
// ENTITY OF CAMERA & AI SURVEILLANCE MODULE
// ============================================================================
Table cameras {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_code VARCHAR(50) [not null, unique, note: 'CAM-001']
  name VARCHAR(100) [not null, note: 'Tên camera']
  mounting_height NUMERIC(5,2)
  orientation NUMERIC(5,2)
  tilt_angle NUMERIC(5,2)
  status VARCHAR(50) [not null, note: 'ACTIVE | DECOMMISSIONED']
  operational_status VARCHAR(50) [not null, note: 'ONLINE | OFFLINE | ERROR']
  installed_at TIMESTAMPTZ
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ
}

Table camera_specifications {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, unique, note: 'FK to cameras']
  manufacturer VARCHAR(100)
  model VARCHAR(100)
  serial_number VARCHAR(100)
  resolution VARCHAR(50)
  fps INT
  lens VARCHAR(100)
  focal_length VARCHAR(50)
  field_of_view NUMERIC(5,2)
  night_vision BOOLEAN
  ptz_supported BOOLEAN
  weather_proof BOOLEAN
  firmware_version VARCHAR(50)
}

Table camera_stream_configurations {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, unique, note: 'FK to cameras']
  protocol VARCHAR(50) [not null, note: 'RTSP | RTMP | HTTP | HTTPS']
  host VARCHAR(255) [not null]
  port INT [not null]
  username VARCHAR(100)
  credential_ref VARCHAR(255)
  main_stream_path VARCHAR(512) [not null]
  sub_stream_path VARCHAR(512)
  retries_before_alert INT [not null, default: 3]
  timeout_ms INT [not null, default: 5000]
}

Table camera_ai_configurations {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, unique, note: 'FK to cameras']
  person_detection_enabled BOOLEAN [not null, default: false]
  face_recognition_enabled BOOLEAN [not null, default: false]
  face_match_threshold NUMERIC(3,2) [not null, note: 'ngưỡng tin cậy nhận dạng khuôn mặt']
  inference_fps INT [not null, note: 'số frame AI xử lý trong 1s']
}

Table camera_health_logs {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_id UUID [not null, note: 'FK to cameras']
  status VARCHAR(50) [not null, note: 'ONLINE | OFFLINE | ERROR']
  checked_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  latency_ms INT [note: 'Độ trễ camera phản hồi tính bằng mili giây']
  fps NUMERIC(5,2) [note: 'Số frame/s thực tế']
  error_code VARCHAR(100) [note: 'Mã lỗi camera']
  error_message VARCHAR(255) [note: 'Mô tả lỗi camera']
}

// Bảng trung gian Area N-N Camera
Table area_cameras {
  area_id UUID [not null, note: 'FK to areas']
  camera_id UUID [not null, note: 'FK to cameras']
  
  indexes {
    (area_id, camera_id) [pk]
  }
}

// ============================================================================
// ENTITY OF ACCESS REQUEST & SCHEDULING MODULE
// ============================================================================
Table access_requests {
  id UUID [pk, default: `gen_random_uuid()`]
  area_id UUID [not null, note: 'FK to areas — khu vực cần truy cập']
  requester_id UUID [not null, note: 'FK to users — người tạo request (cá nhân hoặc đại diện nhóm)']
  request_type VARCHAR(20) [not null, note: 'INDIVIDUAL | GROUP']
  purpose TEXT [not null, note: 'Mục đích sử dụng']
  start_time TIMESTAMPTZ [not null, note: 'Thời gian bắt đầu sử dụng']
  end_time TIMESTAMPTZ [not null, note: 'Thời gian kết thúc sử dụng']
  status VARCHAR(20) [not null, default: 'PENDING', note: 'PENDING | APPROVED | REJECTED']
  reviewer_id UUID [note: 'FK to users — FM phê duyệt']
  reviewed_at TIMESTAMPTZ [note: 'Thời điểm FM xử lý']
  rejection_reason TEXT [note: 'Lý do từ chối nếu bị reject']
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

Table access_request_members {
  id UUID [pk, default: `gen_random_uuid()`]
  access_request_id UUID [not null, note: 'FK to access_requests']
  user_id UUID [not null, note: 'FK to users — thành viên trong nhóm được yêu cầu truy cập']

  indexes {
    (access_request_id, user_id) [unique]
  }
}

Table user_area_schedules {
  id UUID [pk, default: `gen_random_uuid()`]
  user_id UUID [not null, note: 'FK to users — nhân viên được gán lịch']
  area_id UUID [not null, note: 'FK to areas — khu vực làm việc/truy cập']
  name VARCHAR(150) [not null, note: 'Tên ca/lịch làm việc, vd: Ca sáng, Ca chiều']
  day_of_week VARCHAR(10) [not null, note: 'MONDAY | TUESDAY | WEDNESDAY | THURSDAY | FRIDAY | SATURDAY | SUNDAY']
  start_time TIME [not null, note: 'Giờ bắt đầu ca']
  end_time TIME [not null, note: 'Giờ kết thúc ca (giờ đóng cửa)']
  is_active BOOLEAN [not null, default: true]
  created_by UUID [note: 'FK to users — FM tạo lịch']
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

// Relationships
Ref: camera_specifications.camera_id - cameras.id
Ref: camera_stream_configurations.camera_id - cameras.id
Ref: camera_ai_configurations.camera_id - cameras.id
Ref: camera_health_logs.camera_id > cameras.id

// Area - Camera N-N Relationships
Ref: area_cameras.area_id > areas.id
Ref: area_cameras.camera_id > cameras.id

// Access Request & Scheduling Relationships
Ref: access_requests.area_id > areas.id
Ref: access_requests.requester_id > users.id
Ref: access_requests.reviewer_id > users.id

Ref: access_request_members.access_request_id > access_requests.id
Ref: access_request_members.user_id > users.id

Ref: user_area_schedules.user_id > users.id
Ref: user_area_schedules.area_id > areas.id
Ref: user_area_schedules.created_by > users.id

// ---- LUỒNG 1: ADMIN CONFIGURATION ----

Table areas {
  id UUID [pk, default: `gen_random_uuid()`]
  name VARCHAR(100) [not null]
  area_level INT [not null, default: 1, note: 'Phân cấp bảo mật: >= 1 (1: Public, 2: Semi-Private, 3: Private...)']
  description TEXT
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ [note: 'Soft-delete']
}

Table area_escalation_policies {
  area_id UUID [pk, ref: - areas.id]
  allow_outsourced_guard BOOLEAN [not null, default: true]
  severity_threshold_for_internal VARCHAR(50) [not null, default: 'HIGH', note: 'LOW | MEDIUM | HIGH']
  preferred_guard_type VARCHAR(50) [not null, default: 'ANY', note: 'INTERNAL_ONLY | INTERNAL_PREFERRED | OUTSOURCED_PREFERRED | ANY']
  sla_seconds INT [not null, default: 300]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

Table cameras {
  id UUID [pk, default: `gen_random_uuid()`]
  camera_name VARCHAR(100) [not null]
  rtsp_url VARCHAR(255) [not null, unique]
  area_id UUID [not null, ref: > areas.id]
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ [note: 'Soft-delete']
}

Table users {
  id UUID [pk, default: `gen_random_uuid()`]
  full_name VARCHAR(100) [not null]
  user_code VARCHAR(50) [not null, unique, note: 'MSSV hoặc MSNV']
  role_type VARCHAR(50) [not null, default: 'STUDENT', note: 'ADMIN | FACILITY_MANAGER | INTERNAL_GUARD | OUTSOURCED_GUARD | TEACHER | STUDENT | PRINCIPAL | JANITOR | STAFF']
  can_login BOOLEAN [not null, default: true, note: 'Mặc định TRUE cho tất cả. FALSE = admin khóa tài khoản cá nhân']
  phone_number VARCHAR(20)
  email VARCHAR(255)
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  deleted_at TIMESTAMPTZ [note: 'Soft-delete']
}

Table user_face_data {
  id UUID [pk, default: `gen_random_uuid()`]
  user_id UUID [not null, unique, ref: - users.id]
  image_front_url VARCHAR(255) [not null, note: 'Ảnh chính diện - MinIO']
  image_left_url VARCHAR(255) [not null, note: 'Ảnh góc trái - MinIO']
  image_right_url VARCHAR(255) [not null, note: 'Ảnh góc phải - MinIO']
  embedding_front "vector(512)" [not null, note: 'InsightFace embedding']
  embedding_left "vector(512)" [not null]
  embedding_right "vector(512)" [not null]
  ocr_card_data JSONB [note: 'OCR từ thẻ cứng']
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

Table access_permissions {
  id UUID [pk, default: `gen_random_uuid()`]
  user_id UUID [not null, ref: > users.id]
  area_id UUID [not null, ref: > areas.id]
  valid_from TIMESTAMPTZ [not null]
  valid_to TIMESTAMPTZ [note: 'NULL = vô thời hạn']
  granted_by UUID [not null, ref: > users.id, note: 'Admin đã cấp']
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]

  Note: 'Exclusion Constraint chống overlap thời gian. REVOKE = set valid_to = NOW()'
}

// ---- LUỒNG 3: ROUTING & SCHEDULING ----

Table shifts {
  id UUID [pk, default: `gen_random_uuid()`]
  shift_name VARCHAR(50) [not null, note: 'Ca Sáng, Ca Chiều, Ca Đêm']
  start_time TIME [not null]
  end_time TIME [not null]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
}

Table guard_shifts {
  id UUID [pk, default: `gen_random_uuid()`]
  user_id UUID [not null, ref: > users.id]
  shift_id UUID [not null, ref: > shifts.id]
  work_date DATE [not null]
  assigned_area_id UUID [ref: > areas.id]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]

  indexes {
    (user_id, shift_id, work_date) [unique, name: 'unique_guard_shift']
  }
}

// ---- LUỒNG 2: AI DETECTION ENGINE ----

Table detection_events {
  id UUID [pk, note: 'DO AI ENGINE SINH - không dùng gen_random_uuid - đảm bảo idempotency khi Kafka redeliver']
  camera_id UUID [not null, ref: > cameras.id]
  matched_identity_id UUID [ref: > users.id, note: 'NULL nếu không nhận diện được']
  incident_id UUID [ref: > security_incidents.id, note: 'NULL khi chưa gán vào incident - FK thêm bằng ALTER TABLE do circular dependency']
  track_id VARCHAR(100) [not null, note: 'ByteTrack/DeepSORT tracking ID']
  face_crop_url VARCHAR(255) [note: 'Ảnh cắt khuôn mặt - MinIO']
  full_frame_url VARCHAR(255) [not null, note: 'Ảnh toàn cảnh - MinIO']
  confidence_score FLOAT [note: 'CHECK >= 0 AND <= 1']
  status VARCHAR(50) [not null, note: 'AUTHORIZED | UNAUTHORIZED']
  reason VARCHAR(50) [not null, note: 'VALID_ACCESS | NO_FACE_DETECTED | FACE_NOT_MATCHED | PERMISSION_EXPIRED | LOITERING']
  detected_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]
}

// ---- LUỒNG 3: INCIDENT & ESCALATION ----

Table security_incidents {
  id UUID [pk, default: `gen_random_uuid()`]
  track_id VARCHAR(100) [not null, note: 'Gộp chuỗi xâm nhập cùng track_id']
  first_detection_event_id UUID [not null, ref: > detection_events.id]
  last_detection_event_id UUID [not null, ref: > detection_events.id]
  area_id UUID [not null, ref: > areas.id]
  violation_type VARCHAR(50) [not null, note: 'UNAUTHORIZED_ACCESS | UNKNOWN_PERSON | LOITERING']
  severity VARCHAR(50) [not null, default: 'MEDIUM', note: 'LOW | MEDIUM | HIGH - tra từ severity_rules']
  status VARCHAR(50) [not null, default: 'PENDING', note: 'PENDING | VERIFIED | DISMISSED | RESOLVED']
  is_escalated BOOLEAN [not null, default: false]
  escalated_at TIMESTAMPTZ
  escalated_to_id UUID [ref: > users.id, note: 'Facility Manager nhận leo thang']
  assigned_guard_id UUID [ref: > users.id, note: 'Bảo vệ xử lý thực tế']
  sla_deadline TIMESTAMPTZ [not null]
  acknowledged_at TIMESTAMPTZ [note: 'NULL = chưa tiếp nhận']
  dismissal_note TEXT [note: 'Bắt buộc khi DISMISSED - cho AI retrain']
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  resolved_at TIMESTAMPTZ
}

// ---- LUỒNG 4: AUDIT LOGS ----

Table audit_logs {
  id UUID [pk, default: `gen_random_uuid()`]
  actor_id UUID [ref: > users.id, note: 'NULL nếu hệ thống tự động']
  action_type VARCHAR(100) [not null, note: 'CONFIG_AREA, VERIFY_INCIDENT, AUTO_ESCALATE, EXPORT_REPORT...']
  old_data JSONB
  new_data JSONB
  description TEXT [not null]
  ip_address VARCHAR(45)
  created_at TIMESTAMPTZ [not null, default: `CURRENT_TIMESTAMP`]

  Note: 'APPEND-ONLY - Trigger chặn UPDATE/DELETE'
}

// ---- CONFIG TABLES ----

Table severity_rules {
  id UUID [pk, default: `gen_random_uuid()`]
  violation_type VARCHAR(50) [not null, note: 'UNAUTHORIZED_ACCESS | UNKNOWN_PERSON | LOITERING']
  area_level INT [not null, note: 'Khớp chính xác với areas.area_level']
  severity VARCHAR(50) [not null, note: 'LOW | MEDIUM | HIGH']
  is_active BOOLEAN [not null, default: true]
  created_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]

  indexes {
    (violation_type, area_level) [unique, name: 'unique_severity_rule']
  }
}

Table system_config {
  config_key VARCHAR(100) [pk]
  config_value VARCHAR(500) [not null]
  description TEXT
  updated_at TIMESTAMPTZ [default: `CURRENT_TIMESTAMP`]

  Note: 'Key-value config tổng quát. Validate ở tầng ứng dụng.'
}
